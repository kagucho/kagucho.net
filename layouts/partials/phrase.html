{{- /* 見出しを「、」ごとのまとまりに分けて出力する。 各まとまりは CSS（.ph）で途中改行しにくくなるので、スマホでも 「〜なりま／す。」のような泣き別れが起きず、「、」の後ろで折り返される。 使い方：{{ partial "phrase.html" "未経験でも、なんとかなります。" }} */ -}} {{- $parts := split . "、" -}} {{- $last := sub (len $parts) 1 -}} {{- range $i, $p := $parts -}} {{ $p }}{{ if lt $i $last }}、{{ end }} {{- end -}}
